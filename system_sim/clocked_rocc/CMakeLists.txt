cmake_minimum_required(VERSION 3.16)
project(mlx_clocked_rocc LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../clocked_device clocked-device)
add_subdirectory(../model_image model-image)
add_library(mlx_clocked_rocc STATIC adapter.cc progress.cc)
target_include_directories(mlx_clocked_rocc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_clocked_rocc PUBLIC mlx_system_profile mlx_clocked_device)
target_compile_options(mlx_clocked_rocc PRIVATE -Wall -Wextra -Werror)
add_executable(clocked-rocc-driver ../../tests/clocked_rocc_driver.cc ../physical_device/mapped_memory.cc)
target_link_libraries(clocked-rocc-driver PRIVATE mlx_clocked_rocc)
target_compile_options(clocked-rocc-driver PRIVATE -Wall -Wextra -Werror)
