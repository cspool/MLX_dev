cmake_minimum_required(VERSION 3.16)
project(mlx_native_device LANGUAGES C CXX)
set(MLX_ENABLE_MLIR OFF CACHE BOOL "Device uses the compiled program contract")
add_subdirectory(../../simulator_ext/tagged tagged-core)
add_library(mlx_native_device STATIC device.cc)
set_target_properties(mlx_native_device PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mlx_native_device PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_native_device PUBLIC mlx_tagged)
target_compile_options(mlx_native_device PRIVATE -Wall -Wextra -Werror)
add_executable(mlx-device-test protocol_driver.cc)
target_link_libraries(mlx-device-test PRIVATE mlx_native_device)
enable_testing()
add_executable(mlx-device-lifecycle-test ../../tests/native_device_lifecycle.cc)
target_link_libraries(mlx-device-lifecycle-test PRIVATE mlx_native_device)
target_compile_options(mlx-device-lifecycle-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME mlx-native-device-lifecycle COMMAND mlx-device-lifecycle-test)
