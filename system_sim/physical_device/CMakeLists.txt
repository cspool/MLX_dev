cmake_minimum_required(VERSION 3.16)
project(mlx_physical_device_bridge LANGUAGES CXX)
find_package(OpenSSL REQUIRED)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../../simulator_ext/tensor_model tensor-model)
add_library(mlx_matrix_wire STATIC matrix_wire.cc)
target_sources(mlx_matrix_wire PRIVATE vector_wire.cc)
target_sources(mlx_matrix_wire PRIVATE memory_wire.cc)
target_sources(mlx_matrix_wire PRIVATE pair_wire.cc)
target_sources(mlx_matrix_wire PRIVATE pair_runtime.cc)
target_include_directories(mlx_matrix_wire PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_matrix_wire PUBLIC mlx_tensor_values)
target_compile_options(mlx_matrix_wire PRIVATE -Wall -Wextra -Werror)
add_executable(matrix-wire-dump wire_dump.cc)
target_link_libraries(matrix-wire-dump PRIVATE mlx_matrix_wire)
target_compile_options(matrix-wire-dump PRIVATE -Wall -Wextra -Werror)
add_executable(vector-wire-dump vector_wire_dump.cc)
target_link_libraries(vector-wire-dump PRIVATE mlx_matrix_wire)
target_compile_options(vector-wire-dump PRIVATE -Wall -Wextra -Werror)
add_executable(memory-wire-dump memory_wire_dump.cc)
target_link_libraries(memory-wire-dump PRIVATE mlx_matrix_wire)
target_compile_options(memory-wire-dump PRIVATE -Wall -Wextra -Werror)
add_library(mlx_spike_matrix MODULE spike_matrix.cc)
target_sources(mlx_spike_matrix PRIVATE mapped_memory.cc)
target_sources(mlx_spike_matrix PRIVATE asset_source.cc spike_asset_source.cc)
target_include_directories(mlx_spike_matrix PRIVATE ../../build/chipyard-native/toolchains/riscv-tools/riscv-isa-sim/riscv)
target_link_libraries(mlx_spike_matrix PRIVATE mlx_matrix_wire OpenSSL::Crypto)
target_compile_options(mlx_spike_matrix PRIVATE -Wall -Wextra -Werror)
add_executable(mapped-memory-contract ../../tests/mapped_memory_contract.cc mapped_memory.cc)
target_link_libraries(mapped-memory-contract PRIVATE mlx_matrix_wire)
target_compile_options(mapped-memory-contract PRIVATE -Wall -Wextra -Werror)
add_executable(asset-source-contract ../../tests/asset_source_contract.cc asset_source.cc)
target_link_libraries(asset-source-contract PRIVATE mlx_matrix_wire)
target_compile_options(asset-source-contract PRIVATE -Wall -Wextra -Werror)
