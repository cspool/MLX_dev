cmake_minimum_required(VERSION 3.16)
project(mlx_system_model_image LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT TARGET mlx_clocked_device)
  add_subdirectory(../clocked_device clocked-device)
endif()
add_library(mlx_system_profile STATIC profile.cc)
target_include_directories(mlx_system_profile PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_system_profile PUBLIC mlx_clocked_device)
target_compile_options(mlx_system_profile PRIVATE -Wall -Wextra -Werror)
add_executable(system-profile-dump profile_dump.cc)
target_link_libraries(system-profile-dump PRIVATE mlx_system_profile)
target_compile_options(system-profile-dump PRIVATE -Wall -Wextra -Werror)
