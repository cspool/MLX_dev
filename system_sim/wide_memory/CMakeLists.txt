cmake_minimum_required(VERSION 3.16)
project(mlx_wide_memory LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
find_package(OpenSSL REQUIRED)
set(TESTCHIP ${CMAKE_CURRENT_SOURCE_DIR}/../../build/chipyard-native/generators/testchipip/src/main/resources/testchipip/csrc)
add_library(mlx_wide_memory STATIC memory.cc ${TESTCHIP}/mm.cc)
set_target_properties(mlx_wide_memory PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mlx_wide_memory PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${TESTCHIP})
target_compile_features(mlx_wide_memory PUBLIC cxx_std_17)
target_link_libraries(mlx_wide_memory PUBLIC PkgConfig::JSONCPP OpenSSL::Crypto)
set_source_files_properties(memory.cc PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
add_executable(wide-memory-contract ../../tests/wide_memory_contract.cc)
target_link_libraries(wide-memory-contract PRIVATE mlx_wide_memory)
target_compile_options(wide-memory-contract PRIVATE -Wall -Wextra -Werror)
