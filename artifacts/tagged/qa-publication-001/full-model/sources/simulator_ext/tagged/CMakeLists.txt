cmake_minimum_required(VERSION 3.16)
project(mlx_tagged_simulator LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
add_library(mlx_tagged STATIC program.cc fp16.cc simulator.cc)
set_target_properties(mlx_tagged PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(mlx_tagged PUBLIC cxx_std_17)
target_include_directories(mlx_tagged PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_tagged PUBLIC PkgConfig::JSONCPP)
target_compile_options(mlx_tagged PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_executable(mlx-tagged-sim main.cc)
target_link_libraries(mlx-tagged-sim PRIVATE mlx_tagged)
add_library(mlx_tagged_c SHARED c_api.cc)
target_link_libraries(mlx_tagged_c PRIVATE mlx_tagged)
target_include_directories(mlx_tagged_c PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
option(MLX_SANITIZERS "Enable address/undefined-behavior sanitizers" OFF)
if(MLX_SANITIZERS)
  target_compile_options(mlx_tagged PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mlx_tagged PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(mlx-numeric-test numeric_test.cc)
target_link_libraries(mlx-numeric-test PRIVATE mlx_tagged)
add_test(NAME fp16_contract COMMAND mlx-numeric-test)
option(MLX_ENABLE_MLIR "Build the typed MLIR source frontend" ON)
if(MLX_ENABLE_MLIR)
  find_package(MLIR CONFIG QUIET PATHS /usr/lib/llvm-14/lib/cmake/mlir)
  if(MLIR_FOUND)
    if(NOT LLVM_PACKAGE_VERSION MATCHES "^14\\.")
      message(FATAL_ERROR "The MLX frontend currently targets LLVM/MLIR 14")
    endif()
    add_executable(mlx-mlir-front mlir_frontend.cc)
    target_include_directories(mlx-mlir-front SYSTEM PRIVATE ${MLIR_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
    target_link_libraries(mlx-mlir-front PRIVATE mlx_tagged MLIR LLVM)
  else()
    message(STATUS "MLIR frontend unavailable: install LLVM/MLIR 14 development packages")
  endif()
endif()
