cmake_minimum_required(VERSION 3.16)
project(mlx_tensor_semantics LANGUAGES C CXX)
set(MLX_ENABLE_MLIR OFF CACHE BOOL "Tensor semantics consumes compiled IR")
add_subdirectory(../tagged tagged-core)
add_subdirectory(../shared_array shared-array)
add_library(mlx_tensor_values STATIC tensor.cc kernels.cc matrix_program.cc ../matrix_schedule/matrix_schedule.cc ../vector_model/vector_program.cc ../vector_schedule/vector_schedule.cc ../memory_model/memory_program.cc ../control_model/rv64_leaf.cc ../control_model/control_program.cc)
target_sources(mlx_tensor_values PRIVATE ../model_io/tensor_memory_port.cc ../model_io/address_space_port.cc ../model_io/queued_physical_port.cc)
target_sources(mlx_tensor_values PRIVATE ../model_io/physical_mux.cc)
target_sources(mlx_tensor_values PRIVATE ../model_events/completion_window.cc)
target_sources(mlx_tensor_values PRIVATE ../control_schedule/control_schedule.cc)
target_sources(mlx_tensor_values PRIVATE ../control_model/qa_span.c)
target_include_directories(mlx_tensor_values PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../control_schedule)
target_include_directories(mlx_tensor_values PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../matrix_schedule ${CMAKE_CURRENT_SOURCE_DIR}/../vector_model ${CMAKE_CURRENT_SOURCE_DIR}/../vector_schedule ${CMAKE_CURRENT_SOURCE_DIR}/../memory_model ${CMAKE_CURRENT_SOURCE_DIR}/../control_model)
target_link_libraries(mlx_tensor_values PUBLIC mlx_tagged mlx_shared_array ${CMAKE_DL_LIBS})
target_compile_features(mlx_tensor_values PUBLIC cxx_std_17)
target_compile_options(mlx_tensor_values PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_executable(mlx-tensor-semantics main.cc)
target_link_libraries(mlx-tensor-semantics PRIVATE mlx_tensor_values)
target_compile_options(mlx-tensor-semantics PRIVATE -Wall -Wextra -Werror)
