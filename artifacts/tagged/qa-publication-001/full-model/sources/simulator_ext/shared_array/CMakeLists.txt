cmake_minimum_required(VERSION 3.16)
project(mlx_shared_array LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SHARED_JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
add_library(mlx_shared_array STATIC resources.cc)
target_include_directories(mlx_shared_array PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(mlx_shared_array PUBLIC cxx_std_17)
target_compile_options(mlx_shared_array PRIVATE -Wall -Wextra -Werror)
target_link_libraries(mlx_shared_array PUBLIC PkgConfig::SHARED_JSONCPP)
