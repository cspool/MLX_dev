cmake_minimum_required(VERSION 3.16)
project(mlx_physical_model LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_subdirectory(../model_storage model-storage)
add_library(mlx_physical_model STATIC physical_memory.cc model_executor.cc ready_graph.cc)
target_include_directories(mlx_physical_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_physical_model PUBLIC mlx_model_storage OpenSSL::Crypto)
target_compile_options(mlx_physical_model PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_executable(mlx-physical-model main.cc)
target_link_libraries(mlx-physical-model PRIVATE mlx_physical_model)
target_compile_options(mlx-physical-model PRIVATE -Wall -Wextra -Werror)
add_executable(mlx-ready-graph ready_main.cc)
target_link_libraries(mlx-ready-graph PRIVATE mlx_physical_model)
target_compile_options(mlx-ready-graph PRIVATE -Wall -Wextra -Werror)
add_executable(physical-memory-contract ../../tests/physical_memory_contract.cc)
target_link_libraries(physical-memory-contract PRIVATE mlx_physical_model)
target_compile_options(physical-memory-contract PRIVATE -Wall -Wextra -Werror)

target_link_libraries(physical-mux-contract PRIVATE mlx_physical_model)
target_compile_options(physical-mux-contract PRIVATE -Wall -Wextra -Werror)
add_executable(completion-window-contract ../../tests/completion_window_contract.cc)
target_link_libraries(completion-window-contract PRIVATE mlx_physical_model)
target_compile_options(completion-window-contract PRIVATE -Wall -Wextra -Werror)
add_executable(template-loader-contract ../../tests/template_loader_contract.cc)

add_executable(shared-offer-cache-contract ../../tests/shared_offer_cache_contract.cc)

target_compile_options(shared-offer-cache-contract PRIVATE -Wall -Wextra -Werror)
