cmake_minimum_required(VERSION 3.16)
project(mlx_vector_window LANGUAGES CXX)
add_subdirectory(../tensor_model tensor-model)
add_library(mlx_vector_window INTERFACE)
target_include_directories(mlx_vector_window INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_vector_window INTERFACE mlx_tensor_values)
add_executable(mlx-vector-window main.cc)
target_link_libraries(mlx-vector-window PRIVATE mlx_vector_window)
target_compile_options(mlx-vector-window PRIVATE -Wall -Wextra -Werror)
add_executable(vector-external-memory ../../tests/vector_external_memory.cc)
target_link_libraries(vector-external-memory PRIVATE mlx_vector_window)
target_compile_options(vector-external-memory PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
