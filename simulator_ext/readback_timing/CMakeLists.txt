cmake_minimum_required(VERSION 3.16)
project(mlx_readback_timing LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
add_executable(mlx-readback-timing main.cc)
target_link_libraries(mlx-readback-timing PRIVATE PkgConfig::JSONCPP)
target_compile_features(mlx-readback-timing PRIVATE cxx_std_17)
target_compile_options(mlx-readback-timing PRIVATE -Wall -Wextra -Werror)
