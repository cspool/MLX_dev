cmake_minimum_required(VERSION 3.16)
project(mlx_matrix_window LANGUAGES CXX)
add_subdirectory(../tensor_model tensor-model)
add_library(mlx_matrix_window INTERFACE)
target_include_directories(mlx_matrix_window INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mlx_matrix_window INTERFACE mlx_tensor_values)
add_executable(mlx-matrix-window main.cc)
target_link_libraries(mlx-matrix-window PRIVATE mlx_matrix_window)
target_compile_options(mlx-matrix-window PRIVATE -Wall -Wextra -Werror)
add_executable(matrix-external-memory ../../tests/matrix_external_memory.cc)
target_link_libraries(matrix-external-memory PRIVATE mlx_matrix_window)
target_compile_options(matrix-external-memory PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_executable(model-io-contract ../../tests/model_io_contract.cc)
target_link_libraries(model-io-contract PRIVATE mlx_matrix_window)
target_compile_options(model-io-contract PRIVATE -Wall -Wextra -Werror)
